# Define the input and output file paths
$InputPath = "rtsp://172.19.128.5:9090/playback?cameraid=1000273%241%240%243%240&type=2&begintime=1742213808&endtime=1742213908&substream=0&recordType=1"
$OutputPath = "output.mp4"

# Duration in HH:MM:SS.MILLISECONDS format
$Duration = "00:01:40.04"

# Construct the FFmpeg command with the duration
$ffmpegCommand = ".\ffmpeg1 -rtsp_transport tcp -i `"$InputPath`" -c:v libx264 -preset ultrafast -tune zerolatency -movflags +frag_keyframe+empty_moov -t $Duration -f mp4 -y `"$OutputPath`""

$ffprobe = ".\ffprobe -v error -show_entries format=duration -sexagesimal -of default=noprint_wrappers=1:nokey=1 -rtsp_transport tcp -i `"$InputPath`""


# Execute the command
Invoke-Expression $ffmpegCommand
Invoke-Expression $ffprobe
